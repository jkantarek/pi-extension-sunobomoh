# Contract: State Types

**Files**: `src/state/types.ts`, `src/state/read-model.ts`, `src/state/store.ts`, `src/state/query.ts`

Patterns: **Event Sourcing**, **Projection/Read Model**, **CQRS**, **Value Objects**,
**Fluent Builder with Immutable Accumulation**.

---

## Two-Representation Rule

All four state modules maintain a strict boundary between disk format and domain logic:

| Layer | Types | `outcomes` | ids/timestamps |
|---|---|---|---|
| Wire (JSONL) | `StateEntryJson`, etc. | `Record<string, TagOutcomeJson>` | plain `string` |
| In-memory | `StateEntry`, `ReadModel` | `ReadonlyMap<TagId, TagOutcome>` | branded types |

`parseStateEntry(json: StateEntryJson): StateEntry` is the single conversion point.
No other code crosses this boundary.

---

## Wire Types (`src/state/types.ts`)

Discriminated union on `type`. Written to / parsed from JSONL only.
All `id` fields are ULID strings (26-char Crockford base32, lexicographically sortable by time).
ULIDs are generated by `IdFactory` from `src/core/ids.ts`; never by `crypto.randomUUID()`.

```typescript
export type StateLineJson =
  | StateEntryJson
  | StatePatchJson
  | SteeringRunJson
  | SchedulerRunJson;
```

### `StateEntryJson`

```typescript
/**
 * @example
 * ```ts @import.meta.vitest
 * import { isStateEntryJson } from '../state/types.js';
 * const line: unknown = {
 *   type: 'state_entry', id: '01ARZ3NDEKTSV4RRFFQ69G5FAV', sourceId: 'github',
 *   sourceUri: 'github:///owner/repo/issues/1',
 *   timestamp: '2026-05-07T10:00:00Z',
 *   tags: ['needs-review'], outcomes: {},
 *   data: {}, needsAttention: false,
 *   metadata: { watchedAt: '2026-05-07T10:00:00Z' },
 * };
 * expect(isStateEntryJson(line)).toBe(true);
 * ```
 */
export interface StateEntryJson {
  readonly type: 'state_entry';
  readonly id: string;
  readonly sourceId: string;
  readonly sourceUri: string;
  readonly label: string;           // from WatcherDefinition.extractLabel() — TUI display name
  readonly timestamp: string;
  readonly tags: readonly string[];
  readonly outcomes: Readonly<Record<string, TagOutcomeJson>>;
  readonly data: unknown;
  readonly hydratedData?: unknown;
  readonly needsAttention: boolean;
  readonly attentionScore?: number;
  readonly metadata: StateEntryMetadataJson;
}

export interface StateEntryMetadataJson {
  readonly watchedAt: string;
  readonly hydratedAt?: string;
  readonly lastSteeringAt?: string;
  readonly steeringDecision?: 'promoted' | 'demoted' | 'unchanged';
}

export interface TagOutcomeJson {
  readonly tagId: string;
  readonly status: 'pending' | 'active' | 'resolved' | 'dismissed';
  readonly outcome?: unknown;
  readonly resolvedAt?: string;
  readonly notes?: string;
}
```

### `StatePatchJson`

```typescript
/**
 * @example
 * ```ts @import.meta.vitest
 * import { isStatePatchJson } from '../state/types.js';
 * const p: unknown = { type: 'state_patch', id: '01ARZ3NDEKTSV4RRFFQ69G5FBB', targetId: '01ARZ3NDEKTSV4RRFFQ69G5FAV',

export interface StatePatchJson {
  readonly type: 'state_patch';
  readonly id: string;
  readonly targetId: string;
  readonly timestamp: string;
  readonly patch: Partial<Pick<StateEntryJson,
    'needsAttention' | 'attentionScore' | 'outcomes' | 'tags'>>;
  readonly reason?: string;
}
```

## In-Memory Types (`src/state/types.ts` continued)

All ids, timestamps, and URIs are branded. `outcomes` is a `ReadonlyMap`.

```typescript
import type { EntryId, WatcherId, TagId, IsoTimestamp, ResourceUri } from '../core/brands.js';

export interface StateEntry {
  readonly type: 'state_entry';
  readonly id: EntryId;
  readonly sourceId: WatcherId;
  readonly sourceUri: ResourceUri;
  readonly label: string;
  readonly timestamp: IsoTimestamp;
  readonly tags: readonly TagId[];
  readonly outcomes: ReadonlyMap<TagId, TagOutcome>;
  readonly data: unknown;
  readonly hydratedData?: unknown;
  readonly needsAttention: boolean;
  readonly attentionScore?: number;
  readonly metadata: StateEntryMetadata;
}

export interface StateEntryMetadata {
  readonly watchedAt: IsoTimestamp;
  readonly hydratedAt?: IsoTimestamp;
  readonly lastSteeringAt?: IsoTimestamp;
  readonly steeringDecision?: 'promoted' | 'demoted' | 'unchanged';
}

export interface TagOutcome {
  readonly tagId: TagId;
  readonly status: 'pending' | 'active' | 'resolved' | 'dismissed';
  readonly outcome?: unknown;
  readonly resolvedAt?: IsoTimestamp;
  readonly notes?: string;
}

/** Deserialisation boundary — only call this inside StateStore.load(). */
export declare const parseStateEntry: (json: StateEntryJson) => StateEntry;

## ReadModel (`src/state/read-model.ts`)

Pattern: **Projection / Read Model** — built by folding the event log.

```typescript
import type { EntryId, WatcherId, TagId } from '../core/brands.js';

export interface ReadModel {
  readonly byId:           ReadonlyMap<EntryId,   StateEntry>;
  readonly byTag:          ReadonlyMap<TagId,      readonly EntryId[]>;
  readonly bySourceId:     ReadonlyMap<WatcherId,  readonly EntryId[]>;
  readonly needsAttention: ReadonlySet<EntryId>;
  readonly lastSteeringRun?: SteeringRunJson;
  readonly entryCount:     number;
}

/**
 * Initial empty projection.
 *
 * @example
 * ```ts @import.meta.vitest
 * import { emptyModel } from '../state/read-model.js';
 * const m = emptyModel();
 * expect(m.entryCount).toBe(0);
 * expect(m.byId.size).toBe(0);
 * expect(m.needsAttention.size).toBe(0);
 * ```
 */
export declare const emptyModel: () => ReadModel;

/**
 * Pure fold function. StateStore calls: lines.reduce(projectLine, emptyModel()).
 * Each StateLineJson type is handled independently — extensible without modifying callers.
 *
 * @example
 * ```ts @import.meta.vitest
 * import { emptyModel, projectLine } from '../state/read-model.js';
 * import { makeTestEntryJson } from '../state/test-fixtures.js';
 * const entry = makeTestEntryJson({ needsAttention: true });
 * const model = projectLine(emptyModel(), entry);
 * expect(model.entryCount).toBe(1);
 * expect(model.needsAttention.size).toBe(1);
 * ```
 */
export declare const projectLine: (model: ReadModel, line: StateLineJson) => ReadModel;
```

## StateStore Public API (`src/state/store.ts`)

Pattern: **Thin Repository** — append to disk, load from disk, expose projection.
CQRS write-side. Returns `Result` — never throws.

```typescript
import type { Result } from '../core/result.js';
import type { FileSystem, Clock } from '../core/ports.js';

export interface StateStoreAPI {
  /** Append one or more lines. Single write call — atomic at the OS level. */
  append(lines: readonly StateLineJson[]): Promise<Result<void, Error>>;
  /** Read all lines, rebuild ReadModel. Must be called once before any reads. */
  load(): Promise<Result<void, Error>>;
  /** Current projection. Empty until load() resolves successfully. */
  readonly model: Readonly<ReadModel>;
}

/** Factory — inject FileSystem and Clock for test/prod portability. */
export declare const createStateStore: (
  filePath: string,
  fs: FileSystem,
  clock: Clock,
) => StateStoreAPI;
```

## StateQuery Public API (`src/state/query.ts`)

Pattern: **Fluent Builder with Immutable Accumulation** — each method returns a new
`StateQuery` value object (never mutates). Final `.execute()` applies all accumulated
predicates in one `filter` pass.

```typescript
import type { WatcherId, TagId } from '../core/brands.js';

export interface StateQueryAPI {
  byTag(tagId: TagId): StateQueryAPI;
  byWatcher(watcherId: WatcherId): StateQueryAPI;
  needsAttention(value?: boolean): StateQueryAPI;
  since(timestamp: string): StateQueryAPI;
  until(timestamp: string): StateQueryAPI;
  execute(): readonly StateEntry[];
  count(): number;
}

/**
 * @example
 * ```ts @import.meta.vitest
 * import { createStateQuery } from '../state/query.js';
 * import { emptyModel } from '../state/read-model.js';
 * const q = createStateQuery(emptyModel());
 * expect(q.count()).toBe(0);
 * expect(q.needsAttention().count()).toBe(0);
 * ```
 */
export declare const createStateQuery: (model: ReadModel) => StateQueryAPI;
```

Internal implementation uses predicate accumulation:

```typescript
type Predicate = (e: StateEntry) => boolean;
// Each builder method calls:
const addPredicate = (q: StateQuery, p: Predicate): StateQuery =>
  createStateQuery(q.model, [...q.predicates, p]);
// execute() calls:
// model.byId values filtered by predicates.every(p => p(entry))
```
